### 🛑 The "Empty File" Critical Failure

Despite your previous efforts, **two required configuration files are currently empty (0 bytes)**.
Your backend is trying to read these files to send instructions to AWS and Kubernetes, but it finds nothing.

1. **`charts/spot-optimizer-agent/templates/daemonset.yaml`** is **EMPTY**.
* *Result:* The Agent Injector creates a Namespace but deploys *nothing* inside it.


2. **`backend/templates/aws/full-access-role.yaml`** is **EMPTY**.
* *Result:* When you try to add a new AWS account, the CloudFormation stack creates a role with **no permissions**, so discovery fails.

#### Step 3: Hard-Code the Deployment in Python (The "Bulletproof" Fix)

Since reading files from disk is proving error-prone, **embed the logic directly in your python service**.

Update `backend/services/agent_injector.py`. Replace the `_deploy_daemonset` method with this code that creates the objects programmatically. This bypasses the empty file issue entirely.

```python
    def _deploy_daemonset(self, session, cluster_name, region, token, cluster_id, api_key):
        """Deploy the agent using raw Kubernetes API calls (No YAML files needed)"""
        eks = session.client('eks', region_name=region)
        cluster_info = eks.describe_cluster(name=cluster_name)['cluster']
        
        # 1. Setup Connection (SSL Fix)
        with tempfile.NamedTemporaryFile(delete=False) as ca_file:
            ca_file.write(base64.b64decode(cluster_info['certificateAuthority']['data']))
            ca_file.flush()
            
            config = k8s_client.Configuration()
            config.host = cluster_info['endpoint']
            config.ssl_ca_cert = ca_file.name
            config.api_key['authorization'] = f"Bearer {token}"
            api_client = k8s_client.ApiClient(config)
            
            # 2. Define Resources
            core_v1 = k8s_client.CoreV1Api(api_client)
            apps_v1 = k8s_client.AppsV1Api(api_client)
            rbac_v1 = k8s_client.RbacAuthorizationV1Api(api_client)
            
            ns = "spot-optimizer"
            
            # 3. Create Namespace
            try:
                core_v1.create_namespace(k8s_client.V1Namespace(metadata=k8s_client.V1ObjectMeta(name=ns)))
            except k8s_client.rest.ApiException as e:
                if e.status != 409: raise e

            # 4. Create ServiceAccount
            sa = k8s_client.V1ServiceAccount(metadata=k8s_client.V1ObjectMeta(name="spot-agent-sa", namespace=ns))
            try:
                core_v1.create_namespaced_service_account(ns, sa)
            except k8s_client.rest.ApiException as e:
                if e.status != 409: raise e

            # 5. Create ClusterRole (Permissions)
            role = k8s_client.V1ClusterRole(
                metadata=k8s_client.V1ObjectMeta(name="spot-optimizer-role"),
                rules=[
                    k8s_client.V1PolicyRule(api_groups=[""], resources=["nodes", "pods", "pods/eviction"], verbs=["get", "list", "watch", "patch", "create", "delete"]),
                    k8s_client.V1PolicyRule(api_groups=["apps"], resources=["daemonsets", "deployments"], verbs=["get", "list", "watch", "patch"]),
                ]
            )
            try:
                rbac_v1.create_cluster_role(role)
            except k8s_client.rest.ApiException as e:
                if e.status != 409: rbac_v1.patch_cluster_role("spot-optimizer-role", role)

            # 6. Create ClusterRoleBinding
            binding = k8s_client.V1ClusterRoleBinding(
                metadata=k8s_client.V1ObjectMeta(name="spot-optimizer-binding"),
                subjects=[k8s_client.RbacV1Subject(kind="ServiceAccount", name="spot-agent-sa", namespace=ns)],
                role_ref=k8s_client.V1RoleRef(kind="ClusterRole", name="spot-optimizer-role", api_group="rbac.authorization.k8s.io")
            )
            try:
                rbac_v1.create_cluster_role_binding(binding)
            except k8s_client.rest.ApiException as e:
                if e.status != 409: rbac_v1.patch_cluster_role_binding("spot-optimizer-binding", binding)

            # 7. Create DaemonSet (The Agent)
            # IMPORTANT: We use the embedded definition to ensure it works even if files are missing
            ds = k8s_client.V1DaemonSet(
                metadata=k8s_client.V1ObjectMeta(name="spot-agent", namespace=ns),
                spec=k8s_client.V1DaemonSetSpec(
                    selector=k8s_client.V1LabelSelector(match_labels={"app": "spot-agent"}),
                    template=k8s_client.V1PodTemplateSpec(
                        metadata=k8s_client.V1ObjectMeta(labels={"app": "spot-agent"}),
                        spec=k8s_client.V1PodSpec(
                            service_account_name="spot-agent-sa",
                            host_network=True,
                            tolerations=[k8s_client.V1Toleration(operator="Exists")],
                            containers=[
                                k8s_client.V1Container(
                                    name="agent",
                                    image="atharva0608/spot-optimizer-agent:latest",
                                    image_pull_policy="Always",
                                    env=[
                                        k8s_client.V1EnvVar(name="CLUSTER_ID", value=str(cluster_id)),
                                        k8s_client.V1EnvVar(name="API_KEY", value=api_key),
                                        k8s_client.V1EnvVar(name="BACKEND_URL", value=settings.API_V1_STR),
                                        k8s_client.V1EnvVar(name="HOST_PROC", value="/host/proc")
                                    ],
                                    volume_mounts=[k8s_client.V1VolumeMount(name="proc", mount_path="/host/proc", read_only=True)]
                                )
                            ],
                            volumes=[k8s_client.V1Volume(name="proc", host_path=k8s_client.V1HostPathVolumeSource(path="/proc"))]
                        )
                    )
                )
            )
            try:
                apps_v1.create_namespaced_daemon_set(ns, ds)
            except k8s_client.rest.ApiException as e:
                if e.status == 409:
                    apps_v1.patch_namespaced_daemon_set("spot-agent", ns, ds)
                else:
                    raise e
            
            os.unlink(ca_file.name)

```
