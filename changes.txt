### 1. 🏗️ Database & Model Changes

We need to store these rules on the `Team` object.

**File:** `backend/models/team.py`

* **Action:** Add a `governance_config` column. This JSON field will store which actions require approval (e.g., `{"account_connect": true, "instance_delete": false}`).

```python
from sqlalchemy import Column, String, ForeignKey, DateTime, JSON # <--- Import JSON
from sqlalchemy.orm import relationship
from datetime import datetime
import uuid
from backend.models.base import Base

class Team(Base):
    __tablename__ = "teams"

    id = Column(String(36), primary_key=True, default=lambda: str(uuid.uuid4()))
    name = Column(String(100), nullable=False)
    organization_id = Column(String(36), ForeignKey("organizations.id"), nullable=False)
    
    # New Column: Stores specific rules for this team
    # Example: { "CONNECT_ACCOUNT": true, "TERMINATE_INSTANCE": true }
    governance_config = Column(JSON, default={}) 
    
    created_at = Column(DateTime, default=datetime.utcnow)

    members = relationship("User", back_populates="team")
    organization = relationship("Organization", back_populates="teams")

```

### 3. 🎨 Frontend Implementation

We need a UI for the Team Lead to toggle these switches.

**File:** `frontend/src/components/settings/TeamGovernance.jsx` (New Component)

```jsx
import React, { useState, useEffect } from 'react';
import { api } from '../../services/api';
import { Card, Switch, Button } from '../shared';
import toast from 'react-hot-toast';

const ACTION_TYPES = [
  { key: 'CONNECT_ACCOUNT', label: 'Connect AWS Account', desc: 'Require approval when a member adds a new cloud account.' },
  { key: 'TERMINATE_INSTANCE', label: 'Terminate Instance', desc: 'Require approval to delete EC2 instances.' },
  { key: 'DELETE_VOLUME', label: 'Delete Volume', desc: 'Require approval to delete EBS volumes.' },
  { key: 'DELETE_SNAPSHOT', label: 'Delete Snapshot', desc: 'Require approval to delete snapshots.' },
  { key: 'RELEASE_IP', label: 'Release Elastic IP', desc: 'Require approval to release static IPs.' },
];

const TeamGovernance = ({ teamId }) => {
  const [config, setConfig] = useState({});
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    fetchConfig();
  }, [teamId]);

  const fetchConfig = async () => {
    // Assuming GET /teams/{id} returns the governance_config field
    const res = await api.get(`/teams/${teamId}`);
    setConfig(res.data.governance_config || {});
  };

  const handleToggle = (key) => {
    setConfig(prev => ({
      ...prev,
      [key]: !prev[key]
    }));
  };

  const saveChanges = async () => {
    setLoading(true);
    try {
      await api.put(`/teams/${teamId}/governance`, config);
      toast.success("Team policies updated");
    } catch (err) {
      toast.error("Failed to update policies");
    } finally {
      setLoading(false);
    }
  };

  return (
    <Card className="p-6">
      <div className="flex justify-between items-center mb-6">
        <div>
          <h2 className="text-xl font-bold">Team Approval Policies</h2>
          <p className="text-gray-500 text-sm">Select which actions require your approval.</p>
        </div>
        <Button onClick={saveChanges} loading={loading}>Save Policies</Button>
      </div>

      <div className="space-y-6">
        {ACTION_TYPES.map(action => (
          <div key={action.key} className="flex justify-between items-center pb-4 border-b last:border-0">
            <div>
              <h3 className="font-medium text-gray-900">{action.label}</h3>
              <p className="text-sm text-gray-500">{action.desc}</p>
            </div>
            <Switch 
              checked={!!config[action.key]} 
              onChange={() => handleToggle(action.key)} 
            />
          </div>
        ))}
      </div>
    </Card>
  );
};

export default TeamGovernance;

```
