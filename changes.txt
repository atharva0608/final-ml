// ── Helpers ──────────────────────────────────────────────────────────────────
function calcStats(schedules) {
  let sleepHours = 0;
  schedules.filter(s => s.isActive).forEach(s => {
    const sh = parseInt(s.sleepAt.split(":")[0]);
    const wh = parseInt(s.wakeAt.split(":")[0]);
    let daily = sh > wh ? (24 - sh + wh) : (sh === wh && s.sleepAt !== s.wakeAt ? 24 : Math.abs(wh - sh));
    if (s.sleepAt === s.wakeAt) daily = 24;
    sleepHours += daily * s.days.length;
  });
  sleepHours = Math.min(sleepHours, 168);
  const savings = Math.round((sleepHours / 168) * 100);
  const activeCount = schedules.filter(s => s.isActive).length;
  const pausedCount = schedules.filter(s => !s.isActive).length;
  const monthlySaved = schedules.filter(s => s.isActive).reduce((a, s) => a + (s.monthlySavings || 0), 0);
  return { sleepHours, awakeHours: 168 - sleepHours, savings, activeCount, pausedCount, monthlySaved };
}

function detectConflicts(newSched, allSchedules) {
  const conflicts = [];
  const sh = parseInt(newSched.sleepAt.split(":")[0]);
  const wh = parseInt(newSched.wakeAt.split(":")[0]);
  allSchedules.filter(s => s.id !== newSched.id && s.isActive).forEach(existing => {
    const sharedClusters = newSched.clusterIds.filter(id => existing.clusterIds.includes(id));
    if (!sharedClusters.length) return;
    const sharedDays = newSched.days.filter(d => existing.days.includes(d));
    if (!sharedDays.length) return;
    const esh = parseInt(existing.sleepAt.split(":")[0]);
    const ewh = parseInt(existing.wakeAt.split(":")[0]);
    const newHours = new Set();
    for (let h = sh; h !== wh; h = (h + 1) % 24) newHours.add(h);
    const existHours = new Set();
    for (let h = esh; h !== ewh; h = (h + 1) % 24) existHours.add(h);
    const overlap = [...newHours].filter(h => existHours.has(h));
    if (overlap.length > 0) conflicts.push({ schedule: existing, sharedClusters, overlapHours: overlap.length });
  });
  return conflicts;
}

// ── Live Progress Banner ──────────────────────────────────────────────────────

  const [elapsed, setElapsed] = useState(134);
  const [step, setStep] = useState(18);
  const total = 23;

  useEffect(() => {
    const t = setInterval(() => {
      setProgress(p => Math.min(p + 0.35, 99));
      setElapsed(e => e + 1);
      setStep(s => Math.min(s + 0.04, total - 0.01));
    }, 1000);

              <div style={{ fontSize: 10, color: "#94a3b8", marginTop: 2 }}>{h.duration} · {h.strategy}</div>
            </div>
          </div>
        );
      })}
    </div>
  );
}

// ── Main Component ────────────────────────────────────────────────────────────
export default function HibernationManagement() {
  const [schedules, setSchedules] = useState(INITIAL_SCHEDULES);
  const [showForm, setShowForm] = useState(false);
  const [editingSchedule, setEditingSchedule] = useState(null);
  const [toast, setToast] = useState(null);
  const [conflictData, setConflictData] = useState(null);
  const [showProgress, setShowProgress] = useState(true);

  const stats = calcStats(schedules);

  const showToast = (msg, type = "success") => {
    setToast({ msg, type });
    setTimeout(() => setToast(null), 2800);
  };

  const handleSave = (schedule) => {
    const conflicts = detectConflicts(schedule, schedules);
    if (conflicts.length > 0) { setConflictData({ conflicts, pendingSchedule: schedule }); return; }
    commitSave(schedule);
  };

  const commitSave = (schedule) => {
    if (editingSchedule) {
      setSchedules(s => s.map(x => x.id === schedule.id ? schedule : x));
      showToast("Schedule updated");
    } else {
      setSchedules(s => [...s, schedule]);
      showToast("Schedule created");
    }
    setShowForm(false);
    setEditingSchedule(null);
    setConflictData(null);
  };

  const handleConflictResolution = (resolution) => {
    const { pendingSchedule, conflicts } = conflictData;
    if (resolution === "deactivate") {
      setSchedules(s => s.map(x => x.id === conflicts[0].schedule.id ? { ...x, isActive: false } : x));
    }
    commitSave(pendingSchedule);
  };

  const handleToggle = (id) => {
    const s = schedules.find(x => x.id === id);
    setSchedules(prev => prev.map(x => x.id === id ? { ...x, isActive: !x.isActive } : x));
    showToast(`Schedule ${s?.isActive ? "paused" : "resumed"}`);
  };

  const handleEdit = (schedule) => { setEditingSchedule(schedule); setShowForm(true); };
  const handleDelete = (id) => { setSchedules(s => s.filter(x => x.id !== id)); showToast("Schedule deleted", "warning"); };
  const handleNew = () => { setEditingSchedule(null); setShowForm(true); };
  const handleCancel = () => { setShowForm(false); setEditingSchedule(null); };

  return (
    <div style={S.page}>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=DM+Sans:wght@400;500;600;700;800&display=swap');
        * { box-sizing: border-box; margin: 0; padding: 0; }
        input[type=time]::-webkit-calendar-picker-indicator { opacity: 0.4; cursor: pointer; }
        button:hover { opacity: 0.86; }
        @keyframes slideIn { from { transform: translateY(-10px); opacity: 0; } to { transform: translateY(0); opacity: 1; } }
        @keyframes fadeIn { from { opacity: 0; } to { opacity: 1; } }
        @keyframes blink { 0%,100% { opacity: 1; } 50% { opacity: 0.3; } }
      `}</style>

      {/* Conflict Modal */}
      {conflictData && (
        <ConflictModal
          conflicts={conflictData.conflicts}
          newSchedule={conflictData.pendingSchedule}
          onResolve={handleConflictResolution}
          onGoBack={() => setConflictData(null)}
        />
      )}

      {/* Toast */}
      {toast && (
        <div style={{ position: "fixed", top: 20, right: 20, zIndex: 300, padding: "10px 16px", borderRadius: 10, background: toast.type === "warning" ? "#fffbeb" : toast.type === "error" ? "#fef2f2" : "#f0fdf4", border: `1px solid ${toast.type === "warning" ? "#fde68a" : toast.type === "error" ? "#fecaca" : "#bbf7d0"}`, color: toast.type === "warning" ? "#92400e" : toast.type === "error" ? "#991b1b" : "#166534", fontSize: 13, fontWeight: 600, boxShadow: "0 4px 20px rgba(0,0,0,0.1)", animation: "slideIn 0.2s ease", display: "flex", alignItems: "center", gap: 8 }}>
          {toast.type === "success" ? <CheckIcon size={14} /> : <AlertCircleIcon size={14} />}
          {toast.msg}
        </div>
      )}

      {/* Top Bar */}
      <div style={S.topbar}>
        <div style={S.topbarLeft}>
          <div style={{ width: 32, height: 32, borderRadius: 8, background: "linear-gradient(135deg, #6366f1, #8b5cf6)", display: "flex", alignItems: "center", justifyContent: "center", color: "#fff" }}>
            <MoonIcon size={16} />
          </div>
          <div>
            <div style={S.pageTitle}>Hibernation Management</div>
            <div style={S.pageSubtitle}>Automated cluster sleep schedules and cost optimization</div>
          </div>
        </div>
        <div style={S.topbarRight}>
          <button style={S.btnDanger}><SunIcon size={14} /> Emergency Wake All</button>
          <button style={S.btnPrimary} onClick={handleNew}><PlusIcon size={14} /> New Schedule</button>
        </div>
      </div>

      {/* Live Progress Banner */}
      {showProgress && <LiveProgressBanner onDismiss={() => setShowProgress(false)} />}

      {/* Content */}
      <div style={S.content}>
        {/* KPI Strip */}
        <div style={S.kpiGrid}>
          <KPICard label="Saved This Month" value={`$${stats.monthlySaved.toLocaleString()}`} sub="vs. always-on baseline" color="#16a34a" accent="#22c55e" />
          <KPICard label="Sleep Hours / Week" value={`${stats.sleepHours}h`} sub="of 168h total" color="#6366f1" accent="#6366f1" />
          <KPICard label="Est. Savings" value={`${stats.savings}%`} sub="per week vs always-on" color={stats.savings > 30 ? "#22c55e" : "#f59e0b"} accent={stats.savings > 30 ? "#22c55e" : "#f59e0b"} />
          <KPICard label="Schedules" value={stats.activeCount} sub={`${stats.pausedCount} paused`} color="#0f172a" accent="#e2e8f0" />
        </div>

        {/* Three-column layout */}
        <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr 340px", gap: 18, alignItems: "start" }}>

          {/* Col 1: Form + Schedules */}
          <div>
            {showForm && <ScheduleForm initial={editingSchedule} onSave={handleSave} onCancel={handleCancel} />}
            <div style={S.card}>
              <div style={S.cardHeader}>
                <span style={S.cardTitle}>Active Schedules ({schedules.length})</span>
                {!showForm && <button style={{ ...S.btnGhost, fontSize: 11 }} onClick={handleNew}><PlusIcon size={12} />New</button>}
              </div>
              {schedules.length === 0 ? (
                <div style={{ padding: "40px 20px", textAlign: "center" }}>
                  <div style={{ fontSize: 36, marginBottom: 10, opacity: 0.2 }}>🌙</div>
                  <div style={{ fontSize: 14, fontWeight: 700, color: "#0f172a", marginBottom: 6 }}>No Hibernation Schedules</div>
                  <div style={{ fontSize: 12, color: "#64748b", marginBottom: 16 }}>Create your first schedule to start saving costs</div>
                  <button style={S.btnPrimary} onClick={handleNew}><PlusIcon size={13} />Create Schedule</button>
                </div>
              ) : (
                <div style={{ padding: "10px" }}>
                  {schedules.map(s => <ScheduleItem key={s.id} schedule={s} onToggle={handleToggle} onEdit={handleEdit} onDelete={handleDelete} />)}
                </div>
              )}
            </div>
          </div>

          {/* Col 2: Audit + Emergency */}
          <div style={{ display: "flex", flexDirection: "column", gap: 18 }}>
            <AuditHistory />
            <EmergencyControls />
          </div>

          {/* Col 3: Savings Report + Strategy */}
          <div style={{ display: "flex", flexDirection: "column", gap: 18 }}>
            <SavingsReport schedules={schedules} />
            <StrategyReference />
          </div>
        </div>
      </div>
    </div>
  );
}
